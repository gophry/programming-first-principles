Naming & Rebinding

Chained Assignemnt

a = 350
b = a

x = y = 350
f'{id(a) = }, {id(b) = }, {a is b = }, {a == b = }'
# assignment binds a name to an existing object — it doesn't copy
f'{id(x) = }, {id(y) = }, {x is y = }, {x == y = }'
'id(a) = 4426621968, id(b) = 4426621968, a is b = True, a == b = True'
'id(x) = 4426621808, id(y) = 4426621808, x is y = True, x == y = True'

Names have no type

a = 5
a = "now a string"
f'{a = }, {type(a) = }'
"a = 'now a string', type(a) = <class 'str'>"

Rebinding

a = "hi"
b = a
a = a + "!"
f'{a = }, {b = }, {a is b = }'   # a moved on, b didn't follow
"a = 'hi!', b = 'hi', a is b = False"

del and Rebinding

  • del removes a binding, not necessarily the object
x = 5
y = x
del x
f'{y = }'
try: 
  f'{x}'
except NameError as e:
  f'x: {e = }'
'y = 5'
'x: e = NameError("name \'x\' is not defined")'

Address, Pointer, Reference

Address

x = c_int(42)       
f'{x.value = }'            

addr = addressof(x)
f'{addr = }'
'x.value = 42'
'addr = 4427295640'

Pointer

p = pointer(x)
f'{p.contents.value = }'

p.contents.value = 99
f'{x.value = }'
'p.contents.value = 42'
'x.value = 99'

ctypes.POINTER

IntPtr = POINTER(c_int)
f'{IntPtr =}'
f'{type(p) == IntPtr = }'
f'{p.contents.value = }'
"IntPtr =<class '__main__.LP_c_int'>"
'type(p) == IntPtr = True'
'p.contents.value = 99'

Null Pointer

null_ptr = IntPtr()
f'IntPtr(): an instance — NULL pointer'
f'---------------------------------------'
f'{null_ptr}'
f'{bool(null_ptr) = }'
'IntPtr(): an instance — NULL pointer'
'---------------------------------------'
'<__main__.LP_c_int object>'
'bool(null_ptr) = False'

ctypes.cast

f'ctypes.cast(addr, IntPtr)=  an instance — pointer aimed at addr which stores int to which x points'
f'---------------------------------------'
live_ptr = cast(addr, IntPtr)
f'{bool(live_ptr) = }'
f"{live_ptr.contents.value = }"
'ctypes.cast(addr, IntPtr)=  an instance — pointer aimed at addr which stores int to which x points'
'---------------------------------------'
'bool(live_ptr) = True'
'live_ptr.contents.value = 99'

ctypes.cast part 2

p1 = cast(addr, POINTER(c_int))
p2 = cast(p1, c_void_p)
p3 = cast(p2, POINTER(c_int))

f'{addr = }'
f'pointer to int: {p1 =}'
f'generic pointer no type: {p2 =}'
f'generic pointer to int pointer: {p3 =}'
f'p1, p2, p3 all hold the SAME address -- only the label changed each time'
'addr = 4427295640'
'pointer to int: p1 =<__main__.LP_c_int object>'
'generic pointer no type: p2 =c_void_p(4427295640)'
'generic pointer to int pointer: p3 =<__main__.LP_c_int object>'
'p1, p2, p3 all hold the SAME address -- only the label changed each time'

Reuse of Code and Argument

def hello(name): 
    print("hello",name)

hello("welcome")
try:
    hello()
except TypeError as e:
    f'{e = }'
hello welcome
'e = TypeError("hello() missing 1 required positional argument: \'name\'")'

Call by Value

def swap1(p, q):
    t = p 
    p = q
    q = t

x1 = c_int(1)
y1 = c_int(2)
f'{x1 = }, {y1 = }'
swap1(x1.value, y1.value)
f'{x1 = }, {y1 = }'
'x1 = c_int(1), y1 = c_int(2)'
'x1 = c_int(1), y1 = c_int(2)'

Call By Reference

def swap2(p, q):
    t = p.contents.value
    p.contents.value = q.contents.value
    q.contents.value = t

x2 = c_int(1)
y2 = c_int(2)
x2_ptr = pointer(x2)
y2_ptr = pointer(y2)
f'{x2 = }, {y2 = }'
swap2(x2_ptr, y2_ptr)
f'{x2 = }, {y2 = }'
'x2 = c_int(1), y2 = c_int(2)'
'x2 = c_int(2), y2 = c_int(1)'
