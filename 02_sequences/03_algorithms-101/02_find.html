def find(arr, target):
    index = 0
    for value in arr:          # iterator walk
        if value == target:    # equality
            return index       # found
        index = index + 1
    return -1                  # not found

Exported source 
def find_mismatch(a, f0, l0, b, f1, l1, r):
    while f0 != l0 and f1 != l1 and r(a[f0], b[f1]):
        f0 += 1
        f1 += 1
    return f0, f1

    a, first, last, pred
):
Exported source 
def find_adjacent_mismatch(a, first, last, pred):
    if first == last:
        return last
    x = a[first]
    first += 1
    while first != last and pred(x, a[first]):
        x = a[first]
        first += 1
    return first
