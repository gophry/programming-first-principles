is_sorted

is_sorted

def is_sorted(
    a, first, last
):

Returns whether [first, last) is sorted.

scores = [5, 5, 10, 35, 40, 30, 80, 80]

f'{is_sorted_until(scores, 0, len(scores)) = }'
f'{scores[:5] = }'
f'{is_sorted(scores, 0, 5) = }'
f'{is_sorted(scores, 0, len(scores)) = }'

f'{is_sorted_until([5, 5, 10], 0, 3) = }'
'is_sorted_until(scores, 0, len(scores)) = 5'
'scores[:5] = [5, 5, 10, 35, 40]'
'is_sorted(scores, 0, 5) = True'
'is_sorted(scores, 0, len(scores)) = False'
'is_sorted_until([5, 5, 10], 0, 3) = 3'
raw = [40, 10, 35, 5, 80, 75, 5, 80]
f'{is_sorted_until(raw, 0, len(raw)) = }'
f'{is_sorted(raw, 0, len(raw)) = }'

f'{is_sorted_until(scores, 4, 4) = }'
f'{is_sorted(scores, 4, 4) = }'
f'{is_sorted_until(scores, 4, 5) = }'
'is_sorted_until(raw, 0, len(raw)) = 1'
'is_sorted(raw, 0, len(raw)) = False'
'is_sorted_until(scores, 4, 4) = 4'
'is_sorted(scores, 4, 4) = True'
'is_sorted_until(scores, 4, 5) = 5'

lower_bound

lower_bound

def lower_bound(
    a, first, last, value
):

Returns the first index in [first, last) where value may be inserted.

scores = [5, 5, 10, 30, 35, 40, 80, 80]
f'{scores = }'
f'{lower_bound(scores, 0, len(scores), 5) = }'
f'{lower_bound(scores, 0, len(scores), 30) = }'
f'{lower_bound(scores, 0, len(scores), 20) = }'
f'{lower_bound(scores, 0, len(scores), 80) = }'
'scores = [5, 5, 10, 30, 35, 40, 80, 80]'
'lower_bound(scores, 0, len(scores), 5) = 0'
'lower_bound(scores, 0, len(scores), 30) = 3'
'lower_bound(scores, 0, len(scores), 20) = 3'
'lower_bound(scores, 0, len(scores), 80) = 6'

binary_search

def binary_search(
    a, first, last, value
):

Returns whether [first, last) contains a value equivalent to value.

scores = [5, 5, 10, 30, 35, 40, 80, 80]
f'{scores = }'

f'{binary_search(scores, 0, len(scores), 5) = }'    # True
f'{binary_search(scores, 0, len(scores), 30) = }'   # True
f'{binary_search(scores, 0, len(scores), 20) = }'   # False
f'{binary_search(scores, 0, len(scores), 80) = }'   # True
'scores = [5, 5, 10, 30, 35, 40, 80, 80]'
'binary_search(scores, 0, len(scores), 5) = True'
'binary_search(scores, 0, len(scores), 30) = True'
'binary_search(scores, 0, len(scores), 20) = False'
'binary_search(scores, 0, len(scores), 80) = True'
