lower_bound

  • lower_bound stops on the first element greater than or equal to the value.

lower_bound

def lower_bound(
    a, first, last, value
):

Returns the first index in [first, last) where value may be inserted.

scores_lb = sorted(list(scores_out_of_100))
f'{scores_lb = }'
f'{lower_bound(scores_lb, 0, len(scores_lb), 5) = }'
f'{lower_bound(scores_lb, 0, len(scores_lb), 30) = }'
f'{lower_bound(scores_lb, 0, len(scores_lb), 40) = }'
f'{lower_bound(scores_lb, 0, len(scores_lb), 80) = }'
'scores_lb = [5, 5, 10, 35, 40, 75, 80, 80]'
'lower_bound(scores_lb, 0, len(scores_lb), 5) = 0'
'lower_bound(scores_lb, 0, len(scores_lb), 30) = 3'
'lower_bound(scores_lb, 0, len(scores_lb), 40) = 4'
'lower_bound(scores_lb, 0, len(scores_lb), 80) = 6'
scores_lb = sorted(list(scores_out_of_100))
lb = Cost(repeats=1)
for value, shape in ((5, '5'), (30, '30'), (40, '40'), (80, '80')):

upper_bound

  • upper_bound stops on the first element strictly greater than it.

upper_bound

def upper_bound(
    a, first, last, value
):

Returns the first index in [first, last) after value.

scores_ub = sorted(list(scores_out_of_100))
f'{scores_ub = }'

f'{upper_bound(scores_ub, 0, len(scores_ub), 5) = }'
f'{upper_bound(scores_ub, 0, len(scores_ub), 30) = }'
f'{upper_bound(scores_ub, 0, len(scores_ub), 40) = }'
f'{upper_bound(scores_ub, 0, len(scores_ub), 80) = }'
'scores_ub = [5, 5, 10, 35, 40, 75, 80, 80]'
'upper_bound(scores_ub, 0, len(scores_ub), 5) = 2'
'upper_bound(scores_ub, 0, len(scores_ub), 30) = 3'
'upper_bound(scores_ub, 0, len(scores_ub), 40) = 5'
'upper_bound(scores_ub, 0, len(scores_ub), 80) = 8'
scores_ub = [5, 5, 10, 30, 35, 40, 80, 80]

def upper_bound_counted(a, first, last, value):
    if first != last and isinstance(a[first], Cost._Item):
        value = Cost._Item(value, a[first].counts)
    return upper_bound(a, first, last, value)

ub = Cost(repeats=1)
for value, shape in ((5, '5'), (30, '30'), (40, '40'), (80, '80')):
    _ = ub.run(upper_bound_counted, scores_ub, 0, len(scores_ub), value,
               name='upper_bound', shape=shape,
               operations=('comparisons', 'ms'))
_ = ub.run(upper_bound_counted, scores_ub, 4, 4, 30,
           name='upper_bound', shape='empty',
           operations=('comparisons', 'ms'))

def binary_search(
    a, first, last, value
