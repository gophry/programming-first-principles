def c_min(seq):
  #Iterator to walk over the sequence
  it = iter(seq)
  # Assume first element as min
  winner = next(it)
  # walk over rest to compare
  for val in it:
    if val < winner:
      winner = val
  return winner

def c_max(seq):
  #Iterator to walk over the sequence
  it = iter(seq)
  # Assume first element as min
  winner = next(it)
  # walk over rest to compare
  for val in it:
    if not val < winner:
      winner = val
  return winner

from programming_first_principles.predicates import sums_to
def find_2sum(a, T, pred=sums_to):
    for x in a:
        for y in a:
            if pred(x, y, T=T):
                return x, y
    return None, None

from programming_first_principles.predicates import sums_to
def find_3sum(a, T, pred=sums_to):
    for x in a:
        for y in a:
            for z in a:
                if pred(x, y, z, T=T):

    return None, None, None
